aws eks update-kubeconfig --name dev-cluster

create argo cd namespace -- kubectl create namespace argocd

Install -- kubectl apply -n argocd -f https://raw.githubusercontent.com/argoproj/argo-cd/stable/manifests/install.yaml

check  -- kubectl get pods -n argocd

wait untill all are running

Expose locally -- kubectl port-forward svc/argocd-server -n argocd 8080:443

Get password: -- $pwd = kubectl get secret argocd-initial-admin-secret `
					-n argocd `
					-o jsonpath="{.data.password}"

					[Text.Encoding]::UTF8.GetString(
						[Convert]::FromBase64String($pwd)
					)
					
use admin username and this password to login to localhost:8080 argo UI
